OtpApiService.kt (Retrofit Interface) 
// Step 1: Add Retrofit to app/build.gradle:
// implementation 'com.squareup.retrofit2:retrofit:2.9.0'
// implementation 'com.squareup.retrofit2:converter-gson:2.9.0'

package com.example.omniotp

import retrofit2.Response
import retrofit2.Retrofit
import retrofit2.converter.gson.GsonConverterFactory
import retrofit2.http.Body
import retrofit2.http.Header
import retrofit2.http.POST

data class SendOtpRequest(val phone: String, val length: Int = 6, val app_name: String = "My App")
data class SendOtpResponse(val success: Boolean, val session_id: String?, val message: String?, val error: String?)
data class VerifyOtpRequest(val session_id: String, val otp: String)
data class VerifyOtpResponse(val success: Boolean, val verified: Boolean, val message: String?, val error: String?)

interface OtpApiService {
    @POST("api/v1/otp/send")
    suspend fun sendOtp(@Header("x-api-key") apiKey: String, @Body request: SendOtpRequest): Response<SendOtpResponse>

    @POST("api/v1/otp/verify")
    suspend fun verifyOtp(@Header("x-api-key") apiKey: String, @Body request: VerifyOtpRequest): Response<VerifyOtpResponse>

    companion object {
        fun create(): OtpApiService = Retrofit.Builder()
            .baseUrl("http://10.0.2.2:5000/") // Change to your public server or Cloudflare URL
            .addConverterFactory(GsonConverterFactory.create())
            .build().create(OtpApiService::class.java)
    }
}
OtpVerificationActivity.kt (Complete Logic) 
// Dispatch OTP & Verify in Kotlin Activity:
val apiKey = "sk_live_omni_7a9f4c82b1d3e6"
val api = OtpApiService.create()

// 1. Send OTP
lifecycleScope.launch {
    val res = api.sendOtp(apiKey, SendOtpRequest(phone = "+919876543210", app_name = "My Android App"))
    if (res.isSuccessful && res.body()?.success == true) {
        val sessionId = res.body()?.session_id
        Toast.makeText(this@OtpActivity, "OTP sent to your phone!", Toast.LENGTH_SHORT).show()
    }
}

// 2. Verify Code
lifecycleScope.launch {
    val res = api.verifyOtp(apiKey, VerifyOtpRequest(session_id = sessionId!!, otp = "123456"))
    if (res.isSuccessful && res.body()?.verified == true) {
        Toast.makeText(this@OtpActivity, "Phone Verified Successfully!", Toast.LENGTH_LONG).show()
    }
}

<script>
const API_URL = "http://localhost:5000"; // Or your Cloudflare Worker URL
const API_KEY = "sk_live_omni_7a9f4c82b1d3e6";
let activeSession = null;

async function sendMobileOTP() {
  const phone = document.getElementById('user-phone').value;
  const res = await fetch(`${API_URL}/api/v1/otp/send`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'x-api-key': API_KEY },
    body: JSON.stringify({ phone, app_name: 'My Website' })
  });
  const data = await res.json();
  if (data.success) {
    activeSession = data.session_id;
    document.getElementById('verify-area').style.display = 'block';
    alert('OTP sent via SMS!');
  }
}

async function verifyMobileOTP() {
  const otp = document.getElementById('user-code').value;
  const res = await fetch(`${API_URL}/api/v1/otp/verify`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'x-api-key': API_KEY },
    body: JSON.stringify({ session_id: activeSession, otp })
  });
  const data = await res.json();
  if (data.verified) alert('Verification successful! Logged in.');
  else alert('Invalid OTP code');
}
</script>
